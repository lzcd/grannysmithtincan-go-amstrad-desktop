include(GNUInstallDirs)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ADWAITA REQUIRED IMPORTED_TARGET libadwaita-1>=1.2 gtk4>=4.8)
pkg_search_module(SDL2 REQUIRED IMPORTED_TARGET sdl2 sdl2-compat)

set(CPC_DESKTOP_RUNTIME "" CACHE PATH "FujiNet-PC RS232 distribution to install")
set(CPC_DESKTOP_ROM "" CACHE FILEPATH "Optional combined CPC 464 OS and BASIC ROM to install")
set(CPC_APP_ID "online.fujinet.go.amstrad.gnome")
set(cpc_runtime_install "${CMAKE_INSTALL_LIBEXECDIR}/fujinet-go-amstrad")
set(cpc_data_install "${CMAKE_INSTALL_DATADIR}/fujinet-go-amstrad")
add_executable(fujinet-go-amstrad main.c debugger.c runtime.c)
target_link_libraries(fujinet-go-amstrad PRIVATE cpc_core PkgConfig::ADWAITA PkgConfig::SDL2)
target_compile_definitions(fujinet-go-amstrad PRIVATE
    CPC_INSTALLED_RUNTIME="${CMAKE_INSTALL_FULL_LIBEXECDIR}/fujinet-go-amstrad"
    CPC_INSTALLED_ROM="${CMAKE_INSTALL_FULL_DATADIR}/fujinet-go-amstrad/cpc464.rom")
install(TARGETS fujinet-go-amstrad DESTINATION "${CMAKE_INSTALL_BINDIR}")
set(metadata "${PROJECT_SOURCE_DIR}/data")
install(FILES "${metadata}/${CPC_APP_ID}.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
install(FILES "${metadata}/${CPC_APP_ID}.metainfo.xml" DESTINATION "${CMAKE_INSTALL_DATADIR}/metainfo")
install(FILES "${metadata}/${CPC_APP_ID}.svg"
    DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps")
if(CPC_DESKTOP_RUNTIME)
    install(PROGRAMS "${CPC_DESKTOP_RUNTIME}/fujinet" DESTINATION "${cpc_runtime_install}")
    install(FILES "${CPC_DESKTOP_RUNTIME}/fnconfig.ini" DESTINATION "${cpc_runtime_install}")
    install(DIRECTORY "${CPC_DESKTOP_RUNTIME}/data" DESTINATION "${cpc_runtime_install}")
endif()
if(CPC_DESKTOP_ROM)
    file(SIZE "${CPC_DESKTOP_ROM}" cpc_rom_size)
    if(NOT cpc_rom_size EQUAL 32768)
        message(FATAL_ERROR "CPC_DESKTOP_ROM must be one combined 32 KiB CPC 464 ROM")
    endif()
    install(FILES "${CPC_DESKTOP_ROM}" DESTINATION "${cpc_data_install}" RENAME cpc464.rom)
endif()
